spec=importlib.util.spec_from_file_location('r093_independent',source)
m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)
results=[]
def check(name,fn):
    fn();results.append({'name':name,'result':'PASS'})
def require(v):
    assert v
def error(fn):
    try:fn()
    except m.ProbeError:return
    raise AssertionError('expected refusal')

# cvf-allow-secret-fixture: constructed synthetic value, never a real credential.

check('plain_known_key_replaced',lambda:require(fake not in m._sanitize_response(json.dumps({'x':fake}).encode(),fake)[0]))
check('inconsistent_usage_unknown',lambda:require(m._usage_record({'usage':{'prompt_tokens':1,'completion_tokens':1,'total_tokens':99}})['status']=='UNKNOWN'))
check('boolean_usage_unknown',lambda:require(m._usage_record({'usage':{'prompt_tokens':True,'completion_tokens':1,'total_tokens':2}})['status']=='UNKNOWN'))
check('zero_usage_known',lambda:require(m._usage_record({'usage':{'prompt_tokens':0,'completion_tokens':0,'total_tokens':0}})['status']=='KNOWN_REPORTED'))

check('wrong_model_refused',lambda:error(lambda:m._validate_local_metadata(m.REGION,'different','unused')))
check('newline_only_normalization',lambda:require(m._normalize_checkout_bytes(b'a\r\nb\r\n')==b'a\nb\n' and m._normalize_checkout_bytes(b'a\rb')==b'a\rb'))
check('committed_seed_currently_verified',lambda:require(m._assert_seed_and_dispatch()['trancheId']==m.TRANCHE_ID))

def owned_timeout():
    start=time.monotonic()
    out,rc,timeout,elapsed=m._run_owned_child([sys.executable,'-B','-c','import time;time.sleep(20)'],b'',start+.35)
    require(timeout and rc is not None and .3<=elapsed<3 and out==b'')
    results.append({'name':'owned_child_timeout_elapsed_observed','result':'PASS','seconds':round(elapsed,3),'returncode':rc})
check('blocked_child_killed_and_reaped',owned_timeout)
def pipe_drain():
    out,rc,timeout,elapsed=m._run_owned_child([sys.executable,'-B','-c','import sys;sys.stdin.buffer.read();sys.stdout.buffer.write(b"X"*1400000)'],b'A'*10000,time.monotonic()+5)
    require(len(out)==1400000 and rc==0 and not timeout)
check('large_pipe_drained_without_join_deadlock',pipe_drain)

def ledger_replay():
    with tempfile.TemporaryDirectory() as t:
        p=pathlib.Path(t)
        c=m._claim_actor_slot('worker',p,'a'*64)
        error(lambda:m._claim_actor_slot('worker',p,'a'*64))
        m._claim_actor_slot('reviewer',p,'a'*64)
        error(lambda:m._claim_actor_slot('reviewer',p,'a'*64))

check('actor_cap_consumed_no_replay',ledger_replay)

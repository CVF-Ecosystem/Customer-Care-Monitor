def hold(reason): raise RuntimeError(reason)
def sha(data): return hashlib.sha256(data).hexdigest()
def no_reparse(path):
    p = Path(path)
    if not p.is_absolute(): hold("ABSOLUTE_PATH_REQUIRED")
    for q in (*reversed(p.parents), p):
        if q.exists() or q.is_symlink():
            a = q.lstat()
            if a.st_file_attributes & stat.FILE_ATTRIBUTE_REPARSE_POINT:
                hold("REPARSE_COMPONENT:" + str(q))
def read(path):
    no_reparse(path)
    return path.read_bytes()
def inside(rel, allowed):
    if not isinstance(rel, str) or rel not in allowed or "\\" in rel or Path(rel).is_absolute() or ".." in Path(rel).parts:
        hold("PATH_NOT_ALLOWED")
    p = ROOT / rel
    no_reparse(p)
    if p.resolve() != p or not p.is_relative_to(ROOT): hold("PATH_ESCAPE_REPARSE")
    return p
def child(argv, seconds=10):
    global CHILD_N
    if ENV is None or LOG_ROOT is None: hold("CHILD_ENV_OR_LOG_NOT_PINNED")
    if OP_END is not None: seconds = min(seconds, OP_END - time.monotonic())
    if seconds <= 0: hold("OPERATION_BUDGET_EXPIRED")
    CHILD_N += 1
    name = f"child_{CHILD_N:03d}"
    begin_utc, begin_mono = datetime.now(timezone.utc).isoformat(), time.monotonic()
    p, outcome, out, err = None, "START_FAILED", b"", b""
    try:
        p = subprocess.Popen(argv, cwd=ROOT, env=ENV, shell=False, stdin=subprocess.DEVNULL,
                             stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        out, err = p.communicate(timeout=seconds)
        outcome = "WAIT_RETURNED"
    except subprocess.TimeoutExpired as timeout_exc:
        out, err = timeout_exc.stdout or b"", timeout_exc.stderr or b""
        outcome = "TIMED_OUT_TERMINATION_ATTEMPTED_WAIT_UNKNOWN"
        p.terminate()
        try:
            out, err = p.communicate(timeout=min(5, max(0, seconds)))
            outcome = "TIMED_OUT_ORIGINAL_WAIT_RETURNED"
        except subprocess.TimeoutExpired:
            outcome = "TIMED_OUT_ORIGINAL_WAIT_UNKNOWN"
            HELD_CHILDREN.append((name, p))
    finally:
        # Owned ignored logs contain only fixed argv and scrubbed environment.
        no_reparse(LOG_ROOT)
        (LOG_ROOT / (name + ".stdout.bin")).write_bytes(out)
        (LOG_ROOT / (name + ".stderr.bin")).write_bytes(err)
        ledger = {"argv": argv, "cwd": str(ROOT), "environment": ENV,
          "startUTC": begin_utc, "startMonotonic": begin_mono,
          "endUTC": datetime.now(timezone.utc).isoformat(), "endMonotonic": time.monotonic(),
          "stdoutBytes": len(out), "stdoutSha256": sha(out),
          "stderrBytes": len(err), "stderrSha256": sha(err),
          "originalWait": outcome, "exitCode": None if p is None else p.returncode}
        put_new(LOG_ROOT / (name + ".json"), ledger)
    if outcome != "WAIT_RETURNED": hold(outcome)
    if p.returncode: hold("COMMAND_EXIT_" + str(p.returncode))
    return out
def git(*args, seconds=10): return child([GIT, *args], seconds)
def clean():
    if git("status", "--porcelain", "--untracked-files=all", "-z"): hold("WORKTREE_DIRTY")
def committed(rel, expected, parse=True):
    p = inside(rel, {rel}); data = read(p)
    if not re.fullmatch(r"[0-9a-f]{64}", expected) or sha(data) != expected: hold("HASH_PIN")
    if git("show", "HEAD:" + rel) != data: hold("NOT_COMMITTED_EXACT_BYTES")
    return json.loads(data) if parse else data
def put_new(path, obj):
    no_reparse(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    no_reparse(path)
    data = (json.dumps(obj, sort_keys=True, separators=(",", ":")) + "\n").encode()
    with path.open("xb") as f: f.write(data); f.flush(); os.fsync(f.fileno())
    return sha(data)
def replace(path, data):
    temp = path.with_name(path.name + ".r139-pending")
    no_reparse(path); no_reparse(temp)
    with temp.open("xb") as f: f.write(data); f.flush(); os.fsync(f.fileno())
    no_reparse(path); no_reparse(temp)
    os.replace(temp, path)
def signal_once(path, obj):
    temp = path.with_name(path.name + ".r139-pending")
    no_reparse(path); no_reparse(temp)
    if path.exists() or temp.exists(): hold("SIGNAL_PATH_OCCUPIED")
    put_new(temp, obj)
    no_reparse(path); no_reparse(temp)
    if path.exists(): hold("SIGNAL_RACE")
    os.replace(temp, path)
def clock_check(start_utc, start_mono):
    if OP_END is not None and time.monotonic() >= OP_END: hold("POST_RESERVATION_60S_EXPIRED")
    elapsed_utc = (datetime.now(timezone.utc) - start_utc).total_seconds()
    elapsed_mono = time.monotonic() - start_mono
    if elapsed_utc < 0 or elapsed_mono < 0 or abs(elapsed_utc - elapsed_mono) > 2:
        hold("CAMPAIGN_CLOCK_DRIFT")

def activate(r, base, pins):
    global OP_END
    clean()
    if git("rev-parse", "HEAD").decode().strip() != base: hold("HEAD_BEFORE_RESERVE")
    nonce = r["nonce"]
    if not isinstance(nonce, str) or not re.fullmatch(r"[0-9a-f]{32}", nonce): hold("NONCE")
    now, mono = datetime.now(timezone.utc), time.monotonic()
    stamp = lambda d: d.strftime("%Y-%m-%dT%H:%M:%S.%fZ")
    start, stop, end = stamp(now), stamp(now + timedelta(seconds=1020)), stamp(now + timedelta(seconds=1200))
    reservation = {"status": "RESERVED_CONSUMED_PENDING_G0_PROOF", "trancheId": "CCMAI-RUNTIME-139",
      "nativeContractId": r["nativeContractId"], "campaignNonce": nonce, "cap": 1, "usedOrUncertain": 1,
      "g0StartUTC": start, "reservedAtUTC": start, "g0Monotonic": mono,
      "monotonicSource": "python.time.monotonic", "pythonPath": pins["tools"]["python"]["path"],
      "hostBootId": r["bootProof"]["hostBootId"], "deadlineUTC": end, "stopRequestedByUTC": stop,
      "stopMonotonic": mono + 1020, "deadlineMonotonic": mono + 1200,
      "cleanupReserveSeconds": 180, "committedDispatchHead": base,
      "freshGrantSha256": r["freshGrant"]["sha256"]}
    OP_END = mono + 60
    reserve_sha = None
    # Reservation creation itself is in the consumed/uncertain failure envelope.
    try:
        reserve_sha = put_new(inside(r["reservationPath"], {r["reservationPath"]}), reservation)
        clock_check(now, mono)
        values = {"{{G0_UTC}}": start, "{{G0_MONO}}": str(mono), "{{STOP_UTC}}": stop,
          "{{DEADLINE_UTC}}": end, "{{RESERVATION_SHA256}}": reserve_sha,
          "{{RESERVATION_PATH}}": r["reservationPath"], "{{BASE_HEAD}}": base, "{{NONCE}}": nonce}
        for rel in CANON:
            clock_check(now, mono)
            text = r["canonicalTemplates"][rel]
            for key, val in values.items(): text = text.replace(key, val)
            if "{{" in text or "}}" in text: hold("UNFILLED_TOKEN")
            replace(inside(rel, ALLOWED), text.encode("utf-8"))
        clock_check(now, mono)
        put_new(inside(RECEIPT, ALLOWED), {"status": "RESERVED_CANONICAL_PENDING_GATES",
          "reservationPath": r["reservationPath"], "reservationSha256": reserve_sha,
          "originalDispatchHead": base, "g0StartUTC": start, "deadlineUTC": end})
        fixed = (
          [pins["tools"]["powershell"]["path"], "-NoProfile", "-NonInteractive", "-ExecutionPolicy", "Bypass", "-File", "scripts/manage_cvf_downstream_catalog.ps1", "-Write", "-ProjectPath", str(ROOT)],
          [sys.executable, "-B", "scripts/cvf_downstream_gate.py", "preflight"],
          [sys.executable, "-B", "scripts/cvf_gates/cvf_downstream_gate_runner.py", "run", "--phase", "pre-commit", "--project-root", ".", "--packet", "docs/work_orders/CCMAI_RUNTIME_139_INTEGRATED_EXECUTION_PROOF.md", "--packet", RECEIPT])
        for argv in fixed: child(argv, OP_END - time.monotonic())
        changed = set(filter(None, git("diff", "--name-only", "-z").decode("utf-8").split("\0")))
        untracked = set(filter(None, git("ls-files", "--others", "--exclude-standard", "-z").decode("utf-8").split("\0")))
        if not (changed | untracked) or (changed | untracked) - ALLOWED: hold("CHANGED_SET")
        child([GIT, "add", "--", *sorted(changed | untracked)], OP_END - time.monotonic())
        staged = set(filter(None, git("diff", "--cached", "--name-only", "-z").decode("utf-8").split("\0")))
        if staged != changed | untracked: hold("STAGED_SET")
        child([GIT, "commit", "-m", "R139 reserve approved integrated G0 campaign"], OP_END - time.monotonic())
        head = git("rev-parse", "HEAD").decode().strip()
        clean()
        clock_check(now, mono)
        if head == base: hold("COMMIT_OR_60S")
        signal_once(inside(r["signalPath"], {r["signalPath"]}),
          {"status": "ROOT_G0_RELEASE", "currentHead": head,
           "originalReservationBaseHead": base, "reservationPath": r["reservationPath"],
           "reservationSha256": reserve_sha, "freshGrantSha256": r["freshGrant"]["sha256"],
           "campaignNonce": nonce, "g0StartUTC": start, "g0Monotonic": mono,
           "stopRequestedByUTC": stop, "deadlineUTC": end,
           "productRuntimeReleased": False, "providerReleased": False})
    except BaseException as exc:
        try: put_new(inside(r["failurePath"], {r["failurePath"]}),
          {"status": "CONSUMED_OR_UNCERTAIN_HOLD_NO_WORKER_SIGNAL", "reservationSha256": reserve_sha,
           "reasonType": type(exc).__name__, "reason": str(exc),
           "observedAtUTC": stamp(datetime.now(timezone.utc))})
        except BaseException: pass
        raise

def main():
    if len(sys.argv) != 4: hold("EXACT_RECIPE_PATH_SHA_AND_HEAD_REQUIRED")
    try:
        activate(*preclock(sys.argv[1], sys.argv[2], sys.argv[3]))
    finally:
        # An unreturned original child prevents normal completion. This wait
        # has NO hard bound and never releases browser/product authority.
        for name, process in HELD_CHILDREN:
            exit_code = process.wait()
            put_new(LOG_ROOT / (name + ".late_wait.json"),
                {"originalWaitEventuallyReturned": True, "exitCode": exit_code,
                 "atUTC": datetime.now(timezone.utc).isoformat()})
if __name__ == "__main__": main()

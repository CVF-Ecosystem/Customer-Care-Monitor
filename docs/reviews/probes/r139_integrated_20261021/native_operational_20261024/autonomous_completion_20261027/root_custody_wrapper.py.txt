def sha(data):
    return hashlib.sha256(data).hexdigest()


def strict_json(raw):
    def unique(pairs):
        out = {}
        for key, value in pairs:
            if key in out:
                raise RuntimeError("DUPLICATE_KEY")
            out[key] = value
        return out
    return json.loads(raw, object_pairs_hook=unique)


def exact_ref(ref, expected_path=None):
    if type(ref) is not dict or set(ref) != {"path", "sha256"}:
        raise RuntimeError("REF_SCHEMA")
    path, expected = ref["path"], ref["sha256"]
    if (type(path) is not str or path.startswith(("/", "\\")) or "\\" in path or
            ":" in path or ".." in Path(path).parts or "." in Path(path).parts or
            re.fullmatch(r"[A-Za-z0-9_./-]+", path) is None):
        raise RuntimeError("REF_PATH")
    if expected_path is not None and path != expected_path:
        raise RuntimeError("REF_PATH_BINDING")
    if type(expected) is not str or re.fullmatch(r"[0-9a-f]{64}", expected) is None:
        raise RuntimeError("REF_SHA_SCHEMA")
    raw = (ROOT / path).read_bytes()
    if sha(raw) != expected:
        raise RuntimeError("REF_SHA_MISMATCH")
    return raw


def write_new(path, obj):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("xb") as file:
        file.write((json.dumps(obj, ensure_ascii=False, indent=2) + "\n").encode("utf-8"))

    result = module.audit_full_twenty(root, pins)

def controller_literal_contracts(source):
    """Read only exact literal dictionaries from the pinned controller preclock."""
    tree = ast.parse(source.decode("utf-8"), filename=DESIGN.as_posix())
    funcs = [node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name == "preclock"]
    if len(funcs) != 1:
        raise RuntimeError("CONTROLLER_PRECLOCK_SCHEMA")
    names = {"source_budget", "required", "source_cap", "accounting", "runtime_cap", "preclock_cap"}
    found = {}
    for node in ast.walk(funcs[0]):
        if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
            name = node.targets[0].id
            if name in names:
                if name in found:
                    raise RuntimeError("CONTROLLER_CONTRACT_DUPLICATE")
                found[name] = ast.literal_eval(node.value)
    if set(found) != names:
        raise RuntimeError("CONTROLLER_CONTRACT_SET")
    return found


def verify_source_preclock_exact_dicts(recipe, build, grant, recovery, controller_source):
    """Catch inherited controller equality holds before a counted original Popen."""
    contract = controller_literal_contracts(controller_source)
    checks = (
        ("SOURCE_BUDGET_RECIPE", recipe.get("freshSourceCheckBudgetOutsideClock"), contract["source_budget"]),
        ("SOURCE_BUDGET_GRANT", grant.get("sourceBudget"), contract["source_budget"]),
        ("EFFECT_BUDGET_RECIPE", recipe.get("newBudget"), contract["required"]),
        ("BOOT_RECOVERY_SOURCE_CAP", recovery.get("sourceBudget"), contract["source_cap"]),
        ("BOOT_RECOVERY_ACCOUNTING", recovery.get("checkAccounting"), contract["accounting"]),
        ("BOOT_RECOVERY_RUNTIME_CAP", recovery.get("unchangedRuntimeCaps"), contract["runtime_cap"]),
        ("BOOT_RECOVERY_PRECLOCK_CAP", recovery.get("preclockInvocation"), contract["preclock_cap"]),
    )
    failures = [name for name, actual, expected in checks if actual != expected]
    if build.get("sourceChecksAccepted") is not True:
        failures.append("SOURCE_CHECK_ACCEPTANCE")
    if failures:
        raise RuntimeError("SOURCE_PRECLOCK_EXACT_DICTS:" + ",".join(failures))
    return {"checked": len(checks) + 1, "controllerSha256": sha(controller_source),
            "sourceBudgetRetained": True, "externalEffectBudgetUnchanged": True}

                      "currentlyReserved": False,
                      "effectAuthority": "OWNERSHIP_METADATA_ONLY_NOT_RUNTIME_RELEASE"}
    if owner != expected_owner or recipe.get("coordinationOwnerSha256") != sha(owner_raw):
        raise RuntimeError("ACTUAL_OWNER_SCHEMA")
    review_ref = recipe.get("autonomousTechnicalReviewRef")
    review_raw = exact_ref(review_ref, TECH_REVIEW)
    if b"Status: REVIEW_PASS" not in review_raw[:400]:
        raise RuntimeError("CURRENT_REVIEW_NOT_ACCEPTED")
    if recipe.get("autonomousTechnicalReleasePath") != V2_AUTHORITY_PATH:
        raise RuntimeError("V2_AUTHORITY_PATH")
    authority = strict_json((ROOT / V2_AUTHORITY_PATH).read_bytes())
    if (authority.get("kind") != "ROOT_R139_AUTONOMOUS_G0_TECHNICAL_RELEASE" or
            authority.get("recipeRef") != {"path": RECIPE.as_posix(), "sha256": recipe_sha} or
            authority.get("independentReviewApproved") is not True or
            authority.get("independentReviewRef") != review_ref or
            authority.get("completionAdmissionRef") != {"path": AUTONOMY_PATH, "sha256": AUTONOMY_SHA} or
            authority.get("acceptedBuildPinsRef") != build_ref or
            authority.get("rootAllocationId") != "R139-AUTONOMOUS-ALLOC-001" or
            authority.get("cumulativeControllerUsedBefore") != expected_accounting["usedBefore"] or
            authority.get("currentAllocatedControllerCeiling") != expected_accounting["totalCap"] or
            authority.get("additionalOriginalInvocationsInThisPacket") != expected_accounting["additionalMax"] or
            any(authority.get("observedNewEffects", {}).get(key) != 0 for key in
                ("controller", "campaign", "services", "cua", "fixture", "key", "provider"))):
        raise RuntimeError("V2_AUTHORITY_CONTRACT")
    verify_actor_bindings(recipe)
    verify_terminal_history(recipe)
    if recipe.get("operatorAvailable") is not True or recipe.get("oldResourceExclusionReviewed") is not True:
        raise RuntimeError("OPERATOR_OR_OLD_SCOPE")
    for path, expected in recipe["oldSha256"].items():
        if sha((ROOT / path).read_bytes()) != expected:
            raise RuntimeError("CANON_DRIFT")
    for path, expected in recipe["protectedUnchangedHashes"]["files"].items():
        if sha((ROOT / path).read_bytes()) != expected:
            raise RuntimeError("PROTECTED_DRIFT")
    if recipe.get("reservationPath") != (READY / "campaign_reservation.json").as_posix():
        raise RuntimeError("RESERVATION_PATH")
    if recipe.get("childLogRoot") != ".cvf/runtime/r139_autonomous_completion_20261028_activation_logs":
        raise RuntimeError("CHILD_LOG_ROOT")
    for path in (recipe["reservationPath"], recipe["signalPath"], recipe["failurePath"], recipe["childLogRoot"]):
        if (ROOT / path).exists():
            raise RuntimeError("FRESH_EFFECT_PATH_OCCUPIED")
    design = (ROOT / DESIGN).read_bytes()
    if sha(design) != DESIGN_SHA or build["pins"]["sourceSha256"].get(DESIGN.as_posix()) != DESIGN_SHA:
        raise RuntimeError("CONTROLLER_SOURCE_PIN")
    grant = strict_json(exact_ref(recipe.get("freshGrant"),
                                  "CVF_SESSION/authority/CCMAI-RUNTIME-139-INTERNAL-CUA-SOURCE-ADMISSION.json"))
    recovery = strict_json(exact_ref(recipe.get("bootRecoveryAdmission"),
                                     "CVF_SESSION/authority/CCMAI-RUNTIME-139-BOOT-IDENTITY-RECOVERY-ADMISSION.json"))
    contract_audit = verify_source_preclock_exact_dicts(recipe, build, grant, recovery, design)
    contract_audit["references"] = verify_source_preclock_refs(recipe, build)
    return build, expected_accounting, contract_audit


def git_read(binary, *args):
    env = {key: os.environ[key] for key in ("SYSTEMROOT", "WINDIR", "TEMP", "TMP", "USERPROFILE",
           "LOCALAPPDATA", "APPDATA", "PROGRAMDATA", "HOMEDRIVE", "HOMEPATH") if key in os.environ}
    env["PATH"] = os.pathsep.join((str(Path(binary).parent), r"C:\Windows\System32"))
    result = subprocess.run([binary, *args], cwd=ROOT, env=env, stdin=subprocess.DEVNULL,
                            capture_output=True, shell=False, timeout=10, check=False)
    if result.returncode != 0:
        raise RuntimeError("GIT_PREFLIGHT")
    return result.stdout


def main():
    if len(sys.argv) != 4 or Path.cwd() != ROOT or not (ROOT / ".cvf/manifest.json").is_file():
        raise RuntimeError("ROOT_ARGS_OR_CWD")
    recipe_path, recipe_sha, dispatch_head = sys.argv[1:]
    if recipe_path != RECIPE.as_posix() or re.fullmatch(r"[0-9a-f]{64}", recipe_sha) is None:
        raise RuntimeError("RECIPE_PATH_OR_SHA")
    raw = (ROOT / recipe_path).read_bytes()
    if sha(raw) != recipe_sha:
        raise RuntimeError("RECIPE_SHA")
    recipe = strict_json(raw)
    if re.fullmatch(r"[0-9a-f]{40}", SOURCE_COMMIT) is None or re.fullmatch(r"[0-9a-f]{32}", NONCE) is None:
        raise RuntimeError("DRAFT_LITERAL_BINDINGS_UNRESOLVED")
    if Path(__file__).read_bytes() != (ROOT / WRAPPER_SOURCE).read_bytes():
        raise RuntimeError("EXTRACTED_WRAPPER_BYTES")
    build, accounting, source_contract_audit = verify_metadata(recipe, recipe_sha)
    guard_ref = recipe.get("delegatedStaticGuardRef")
    if type(guard_ref) is not dict or guard_ref.get("path") != GUARD.as_posix():
        raise RuntimeError("GUARD_REF")
    try:
        first = full_guard(ROOT, build["pins"], GUARD, guard_ref["sha256"])
    except StaticGuardHold as error:
        write_new(ROOT / RUNTIME / "cold_guard_hold.json", error.receipt)
        raise
    write_new(ROOT / RUNTIME / "cold_guard_before_subprocess.json", first)
    write_new(ROOT / RUNTIME / "source_preclock_contract_audit.json", source_contract_audit)
    # The first full 20-file audit is complete before any subprocess, extraction or counted intent.
    python = build["pins"]["tools"]["python"]["path"]
    git_binary = build["pins"]["tools"]["git"]["path"]
    if Path(sys.executable).resolve() != Path(python).resolve():
        raise RuntimeError("PYTHON_EXECUTABLE_PIN")
    if re.fullmatch(r"[0-9a-f]{40}", dispatch_head) is None:
        raise RuntimeError("DISPATCH_HEAD")
    if git_read(git_binary, "rev-parse", "HEAD").decode("ascii").strip() != dispatch_head:
        raise RuntimeError("HEAD_DRIFT")
    if git_read(git_binary, "status", "--porcelain=v1", "--untracked-files=all", "-z"):
        raise RuntimeError("DIRTY_WORKTREE")
    for path in (RECIPE.as_posix(), DESIGN.as_posix(), BUILD_PATH, DELEGATION_PATH, AUTONOMY_PATH,
                 TECH_REVIEW, LEDGER_PATH, V2_AUTHORITY_PATH, WRAPPER_SOURCE.as_posix(), BOOTSTRAP_SOURCE.as_posix()):
        if git_read(git_binary, "show", "HEAD:" + path) != (ROOT / path).read_bytes():
            raise RuntimeError("COMMITTED_BINDING")
    design = (ROOT / DESIGN).read_bytes()
    if sha(design) != DESIGN_SHA or build["pins"]["sourceSha256"].get(DESIGN.as_posix()) != DESIGN_SHA:
        raise RuntimeError("DESIGN_PIN")
    controller_rel = RUNTIME / "activation_bootstrap.py"
    if recipe.get("invocation", {}).get("controllerScriptPath") != controller_rel.as_posix():
        raise RuntimeError("CONTROLLER_PATH")
    controller = ROOT / controller_rel
    if controller.exists():
        raise RuntimeError("CONTROLLER_OCCUPIED")
    custody = ROOT / RUNTIME / "control_outer"
    custody.mkdir(exist_ok=False)
    with controller.open("xb") as file:
        file.write(exact_ref(recipe.get("receiptBindingBootstrapRef"), BOOTSTRAP_SOURCE.as_posix()))
    argv = [python, "-B", str(controller), recipe_path, recipe_sha, dispatch_head]
    def intent():
        write_new(custody / "intent.json", {"kind": "ROOT_ORIGINAL_CONTROL_POPEN_AUTONOMOUS",
                                         "argv": argv, "usedBefore": accounting["usedBefore"],
                                         "usedAtIntent": accounting["usedAtIntent"],
                                         "totalCeiling": accounting["totalCap"], "noAutomaticRetry": True,
                                         "priorPids": [23500, 2560, 28928, 3728],
                                         "createdAtUTC": datetime.now(timezone.utc).isoformat()})
    with (custody / "stdout.bin").open("xb") as out, (custody / "stderr.bin").open("xb") as err:
        process = postcoldgate_popen(ROOT, build["pins"], GUARD, guard_ref["sha256"],
                                    argv, ROOT, out, err,
                                    custody / "cold_guard_before_popen.json", intent)
        write_new(custody / "started.json", {"pid": process.pid, "originalPopenReturned": True,
                                            "startedAtUTC": datetime.now(timezone.utc).isoformat(),
                                            "startedMonotonic": time.monotonic()})
        exit_code = process.wait()
    write_new(custody / "original_wait_exit.json", {"pid": process.pid,
                                                   "originalWaitReturned": True,
                                                   "exitCode": exit_code,
                                                   "returnedAtUTC": datetime.now(timezone.utc).isoformat(),
                                                   "returnedMonotonic": time.monotonic()})
    raise SystemExit(exit_code)


if __name__ == "__main__":
    main()

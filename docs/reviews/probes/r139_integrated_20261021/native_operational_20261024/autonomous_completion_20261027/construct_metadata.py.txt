 if type(value["sha256"]) is not str or re.fullmatch("[0-9a-f]{64}", value["sha256"]) is None: raise ValueError("UNRESOLVED_REF")
 raw = read(path)
 if sha(raw) != value["sha256"]: raise ValueError("REF_HASH")
 return json.loads(raw) if parse else raw

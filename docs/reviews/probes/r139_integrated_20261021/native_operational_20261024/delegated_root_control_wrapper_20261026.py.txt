def sha(data):
    return hashlib.sha256(data).hexdigest()


def strict_json(raw):
    def unique(pairs):
        out = {}
        for key, value in pairs:
            if key in out:
                raise RuntimeError("DUPLICATE_KEY")
            out[key] = value
        return out
    return json.loads(raw, object_pairs_hook=unique)


def exact_ref(ref, expected_path=None):
    if type(ref) is not dict or set(ref) != {"path", "sha256"}:
        raise RuntimeError("REF_SCHEMA")
    path, expected = ref["path"], ref["sha256"]
    if type(path) is not str or path.startswith(("/", "\\")) or "\\" in path or ".." in Path(path).parts:
        raise RuntimeError("REF_PATH")
    if expected_path is not None and path != expected_path:
        raise RuntimeError("REF_PATH_BINDING")
    if type(expected) is not str or re.fullmatch(r"[0-9a-f]{64}", expected) is None:
        raise RuntimeError("REF_SHA_SCHEMA")
    raw = (ROOT / path).read_bytes()
    if sha(raw) != expected:
        raise RuntimeError("REF_SHA_MISMATCH")
    return raw


def write_new(path, obj):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("xb") as file:
        file.write((json.dumps(obj, ensure_ascii=False, indent=2) + "\n").encode("utf-8"))


def full_guard(root, pins, guard_path, guard_sha):
    """Pure complete audit; no subprocess, counter, extraction, or OS identity query."""
    raw = (root / guard_path).read_bytes()
    if sha(raw) != guard_sha:
        raise RuntimeError("GUARD_SOURCE_SHA")
    sys.dont_write_bytecode = True
    module = importlib.machinery.SourceFileLoader("r139_delegated_guard", str(root / guard_path)).load_module()
    result = module.audit_full_twenty(root, pins)
    if result.get("status") != "PASS" or len(result.get("rows", ())) != 20:
        raise StaticGuardHold(result)
    return result


class StaticGuardHold(RuntimeError):
    def __init__(self, receipt):
        super().__init__("FULL20_STATIC_PIN_HOLD")
        self.receipt = receipt


def postcoldgate_popen(root, pins, guard_path, guard_sha, argv, cwd, out, err,
                      audit_receipt_path, intent_writer):
    """The sole launch hook: a fresh full audit precedes intent and Popen."""
    audit = full_guard(root, pins, guard_path, guard_sha)
    write_new(audit_receipt_path, audit)
    intent_writer()
    return subprocess.Popen(argv, cwd=cwd, shell=False, stdin=subprocess.DEVNULL,
                            stdout=out, stderr=err)


def verify_metadata(recipe):
    if recipe.get("kind") != "R139_ROOT_ACTIVATION_RECIPE" or recipe.get("status") != "COMMITTED_PRECLOCK":
        raise RuntimeError("RECIPE_NOT_SEALED")
    if recipe.get("readyForActivation") is not True or recipe.get("rootSealPending") is not False:
        raise RuntimeError("RECIPE_NOT_READY")
    if recipe.get("nonce") != NONCE or recipe.get("nativeContractId") != "r139-cua-boot-identity-v3-20261026":
        raise RuntimeError("CONTRACT_NONCE")
    build_ref = recipe.get("acceptedBuildPinsRef")
    build = strict_json(exact_ref(build_ref, BUILD_PATH))
    if (build.get("kind") != "ROOT_REVIEWED_R139_ACCEPTED_BUILD_PINS" or
            build.get("rootReviewApproved") is not True or build.get("sourceChecksAccepted") is not True or
            build.get("sourceCommit") != SOURCE_COMMIT or build.get("parentAcceptedBuildRef") != PARENT_BUILD or
            build.get("ownerDelegatedAdoptionRef") != {"path": DELEGATION_PATH, "sha256": DELEGATION_SHA}):
        raise RuntimeError("ACCEPTED_BUILD_BINDING")
    delegation_ref = {"path": DELEGATION_PATH, "sha256": DELEGATION_SHA}
    if recipe.get("delegatedAuthorityRef") != delegation_ref:
        raise RuntimeError("DELEGATION_REF")
    delegation = strict_json(exact_ref(delegation_ref))
    if (delegation.get("kind") != "OWNER_DELEGATED_R139_TECHNICAL_BUDGET_AND_QUALIFIED_DOCKER_ADOPTION" or
            delegation.get("ownerApproved") is not True or
            delegation.get("initialAllocation", {}).get("controllersUsed") != 2 or
            delegation.get("initialAllocation", {}).get("controllerTotalCeiling") != 3):
        raise RuntimeError("DELEGATION_BOUNDARY")
    snapshot = delegation["dockerAdoption"]
    docker = build["pins"]["tools"]["docker"]
    if docker != {"path": snapshot["snapshotPath"], "sha256": snapshot["sha256"], "size": snapshot["size"]}:
        raise RuntimeError("DOCKER_SNAPSHOT_BINDING")
    if recipe.get("cumulativeControllerAccounting") != {"usedBefore": 2, "totalCap": 3, "additionalMax": 1, "usedAtIntent": 3}:
        raise RuntimeError("CONTROLLER_ACCOUNTING")
    waits = recipe.get("priorOriginalWaitExitRefs")
    if type(waits) is not list or len(waits) != 2:
        raise RuntimeError("PRIOR_WAITS_SCHEMA")
    for ref, path, pid in zip(waits, PRIOR_WAITS, (23500, 2560)):
        receipt = strict_json(exact_ref(ref, path.as_posix()))
        if receipt.get("pid") != pid or receipt.get("originalWaitReturned") is not True or receipt.get("exitCode") != 1:
            raise RuntimeError("PRIOR_ORIGINAL_NOT_RETAINED")
    owner_ref = recipe.get("delegatedCoordinationOwnerRef")
    owner_raw = exact_ref(owner_ref, OWNER_PUBLIC.as_posix())
    coord = READY / "coordination"
    if recipe.get("coordinationRoot") != coord.as_posix() or not (ROOT / coord).is_dir():
        raise RuntimeError("COORD_PATH")
    if {p.name for p in (ROOT / coord).iterdir()} != {"owner.json"}:
        raise RuntimeError("COORD_EXCLUSIVE")
    if (ROOT / coord / "owner.json").read_bytes() != owner_raw:
        raise RuntimeError("ACTUAL_OWNER_BYTES")
    owner = strict_json(owner_raw)
    expected_owner = {"kind": "R139_ROOT_PRECLOCK_COORDINATION_OWNER", "trancheId": "CCMAI-RUNTIME-139",
                      "nativeContractId": recipe["nativeContractId"], "nonce": NONCE,
                      "owner": "Root original-control custodian", "sourceCommit": SOURCE_COMMIT,
                      "acceptedBuildSha256": build_ref["sha256"], "rootExclusive": True,
                      "freshGrantSha256": recipe["freshGrant"]["sha256"],
                      "delegatedAuthorityRef": delegation_ref, "currentlyReserved": False,
                      "effectAuthority": "OWNERSHIP_METADATA_ONLY_NOT_RUNTIME_RELEASE"}
    if owner != expected_owner or recipe.get("coordinationOwnerSha256") != sha(owner_raw):
        raise RuntimeError("ACTUAL_OWNER_SCHEMA")
    review_ref = recipe.get("delegatedTechnicalReviewRef")
    review_raw = exact_ref(review_ref, TECH_REVIEW)
    if b"Status: REVIEW_PASS" not in review_raw[:400]:
        raise RuntimeError("CURRENT_REVIEW_NOT_ACCEPTED")
    for actor in ("root", "internal", "cuaProducer", "cuaObserver"):
        ack = strict_json(exact_ref(recipe["actorAcks"][actor]))
        if (ack.get("ready") is not True or ack.get("campaignNonce") != NONCE or
                ack.get("sourceCommit") != SOURCE_COMMIT or
                ack.get("acceptedBuildPinsSha256") != build_ref["sha256"] or
                ack.get("campaignReserved") is not False or ack.get("effectsObserved") != 0):
            raise RuntimeError("ACTOR_ACK_BINDING")
    if recipe.get("operatorAvailable") is not True or recipe.get("oldResourceExclusionReviewed") is not True:
        raise RuntimeError("OPERATOR_OR_OLD_SCOPE")
    for path, expected in recipe["oldSha256"].items():
        if sha((ROOT / path).read_bytes()) != expected:
            raise RuntimeError("CANON_DRIFT")
    for path, expected in recipe["protectedUnchangedHashes"]["files"].items():
        if sha((ROOT / path).read_bytes()) != expected:
            raise RuntimeError("PROTECTED_DRIFT")
    if recipe.get("reservationPath") != (READY / "campaign_reservation.json").as_posix():
        raise RuntimeError("RESERVATION_PATH")
    if recipe.get("childLogRoot") != ".cvf/runtime/r139_delegated_reentry_20261026_activation_logs":
        raise RuntimeError("CHILD_LOG_ROOT")
    for path in (recipe["reservationPath"], recipe["signalPath"], recipe["failurePath"], recipe["childLogRoot"]):
        if (ROOT / path).exists():
            raise RuntimeError("FRESH_EFFECT_PATH_OCCUPIED")
    return build


def git_read(binary, *args):
    env = {key: value for key, value in os.environ.items() if not key.upper().startswith("GIT_")}
    result = subprocess.run([binary, *args], cwd=ROOT, env=env, stdin=subprocess.DEVNULL,
                            capture_output=True, shell=False, timeout=10, check=False)
    if result.returncode != 0:
        raise RuntimeError("GIT_PREFLIGHT")
    return result.stdout


def main():
    if len(sys.argv) != 4 or Path.cwd() != ROOT or not (ROOT / ".cvf/manifest.json").is_file():
        raise RuntimeError("ROOT_ARGS_OR_CWD")
    recipe_path, recipe_sha, dispatch_head = sys.argv[1:]
    if recipe_path != RECIPE.as_posix() or re.fullmatch(r"[0-9a-f]{64}", recipe_sha) is None:
        raise RuntimeError("RECIPE_PATH_OR_SHA")
    raw = (ROOT / recipe_path).read_bytes()
    if sha(raw) != recipe_sha:
        raise RuntimeError("RECIPE_SHA")
    recipe = strict_json(raw)
    build = verify_metadata(recipe)
    guard_ref = recipe.get("delegatedStaticGuardRef")
    if type(guard_ref) is not dict or guard_ref.get("path") != GUARD.as_posix():
        raise RuntimeError("GUARD_REF")
    try:
        first = full_guard(ROOT, build["pins"], GUARD, guard_ref["sha256"])
    except StaticGuardHold as error:
        write_new(ROOT / RUNTIME / "cold_guard_hold.json", error.receipt)
        raise
    write_new(ROOT / RUNTIME / "cold_guard_before_subprocess.json", first)
    # The first full 20-file audit is complete before any subprocess, extraction or counted intent.
    python = build["pins"]["tools"]["python"]["path"]
    git_binary = build["pins"]["tools"]["git"]["path"]
    if Path(sys.executable).resolve() != Path(python).resolve():
        raise RuntimeError("PYTHON_EXECUTABLE_PIN")
    if re.fullmatch(r"[0-9a-f]{40}", dispatch_head) is None:
        raise RuntimeError("DISPATCH_HEAD")
    if git_read(git_binary, "rev-parse", "HEAD").decode("ascii").strip() != dispatch_head:
        raise RuntimeError("HEAD_DRIFT")
    if git_read(git_binary, "status", "--porcelain=v1", "--untracked-files=all", "-z"):
        raise RuntimeError("DIRTY_WORKTREE")
    for path in (RECIPE.as_posix(), DESIGN.as_posix(), BUILD_PATH, DELEGATION_PATH, TECH_REVIEW):
        if git_read(git_binary, "show", "HEAD:" + path) != (ROOT / path).read_bytes():
            raise RuntimeError("COMMITTED_BINDING")
    design = (ROOT / DESIGN).read_bytes()
    if sha(design) != DESIGN_SHA or build["pins"]["sourceSha256"].get(DESIGN.as_posix()) != DESIGN_SHA:
        raise RuntimeError("DESIGN_PIN")
    controller_rel = RUNTIME / "activation_controller.py"
    if recipe.get("invocation", {}).get("controllerScriptPath") != controller_rel.as_posix():
        raise RuntimeError("CONTROLLER_PATH")
    controller = ROOT / controller_rel
    if controller.exists():
        raise RuntimeError("CONTROLLER_OCCUPIED")
    custody = ROOT / RUNTIME / "control_outer"
    custody.mkdir(exist_ok=False)
    with controller.open("xb") as file:
        file.write(design)
    argv = [python, "-B", str(controller), recipe_path, recipe_sha, dispatch_head]
    def intent():
        write_new(custody / "intent.json", {"kind": "ROOT_ORIGINAL_CONTROL_POPEN_DELEGATED",
                                         "argv": argv, "usedBefore": 2, "usedAtIntent": 3,
                                         "totalCeiling": 3, "noAutomaticRetry": True,
                                         "priorPids": [23500, 2560],
                                         "createdAtUTC": datetime.now(timezone.utc).isoformat()})
    with (custody / "stdout.bin").open("xb") as out, (custody / "stderr.bin").open("xb") as err:
        process = postcoldgate_popen(ROOT, build["pins"], GUARD, guard_ref["sha256"],
                                    argv, ROOT, out, err,
                                    custody / "cold_guard_before_popen.json", intent)
        write_new(custody / "started.json", {"pid": process.pid, "originalPopenReturned": True,
                                            "startedAtUTC": datetime.now(timezone.utc).isoformat(),
                                            "startedMonotonic": time.monotonic()})
        exit_code = process.wait()
    write_new(custody / "original_wait_exit.json", {"pid": process.pid,
                                                   "originalWaitReturned": True,
                                                   "exitCode": exit_code,
                                                   "returnedAtUTC": datetime.now(timezone.utc).isoformat(),
                                                   "returnedMonotonic": time.monotonic()})
    raise SystemExit(exit_code)


if __name__ == "__main__":
    main()
